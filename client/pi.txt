# 2. Firefox starten und auf URL gehen
URL="http://url"
# Falls Firefox schon läuft, killen
pkill firefox
# Firefox mit Autostart-Option starten
firefox --kiosk "$URL" &

# 3. Mikrofon dauerhaft erlauben
PROFILE=$(ls -d ~/.mozilla/firefox/*.default-release | head -n 1)
if [ -d "$PROFILE" ]; then
  echo 'user_pref("permissions.default.microphone", 1);' >> "$PROFILE/user.js"
  echo 'user_pref("media.navigator.permission.disabled", true);' >> "$PROFILE/user.js"
fi
