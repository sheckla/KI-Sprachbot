from rpi5_ws2812.ws2812 import Color, WS2812SpiDriver
import time

# 60 LEDs, SPI-Bus 0, Device 0 (GPIO10 → Pin 19)
strip = WS2812SpiDriver(spi_bus=0, spi_device=0, led_count=60).get_strip()

while True:
    strip.set_all_pixels(Color(255, 0, 0))  # Rot
    strip.show()
    time.sleep(1)
    strip.set_all_pixels(Color(0, 255, 0))  # Grün
    strip.show()
    time.sleep(1)
    strip.set_all_pixels(Color(0, 0, 255))  # Blau
    strip.show()
    time.sleep(1)

#!/bin/bash
echo "=== 🧩 Raspberry Pi Python/apt Recovery Script ==="
echo "This will repair your system-wide Python after accidental deletion."

# 1️⃣ Clean up broken Python packages
sudo dpkg --remove --force-remove-reinstreq python3 || true
sudo apt --fix-broken install -y || true

# 2️⃣ Reinstall Python core components
sudo apt update -y
sudo apt install --reinstall -y \
    python3-minimal python3 python3-venv python3-full \
    python3-pip python3-setuptools python3-wheel python3-gdbm || true

# 3️⃣ Reinstall APT Python bindings (apt_pkg)
if ! python3 -c "import apt_pkg" 2>/dev/null; then
    echo "⚙️  Reinstalling python3-apt..."
    sudo apt install --reinstall python3-apt -y || {
        echo "Downloading python3-apt manually..."
        ARCH=$(dpkg --print-architecture)
        wget -O /tmp/python3-apt.deb http://archive.raspberrypi.org/debian/pool/main/p/python-apt/python3-apt_2.6.0_${ARCH}.deb
        sudo dpkg -i /tmp/python3-apt.deb
    }
fi

# 4️⃣ Fix dpkg and apt if still broken
sudo apt install --reinstall apt dpkg -y
sudo apt --fix-broken install -y

# 5️⃣ Final system repair and cleanup
sudo apt update -y
sudo apt upgrade -y
sudo apt autoremove -y
sudo apt clean

# 6️⃣ Check versions
echo
echo "✅ Checking restored environment..."
python3 --version 2>/dev/null || echo "❌ Python3 still missing!"
pip3 --version 2>/dev/null || echo "❌ pip3 still missing!"
echo "====================================="
echo "If both versions appear above, your system is fixed."
echo "You can now reinstall any libraries with pip3 install <package>"
