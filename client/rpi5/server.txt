# needs https://pypi.org/project/rpi5-ws2812/#files
from flask import Flask, request, jsonify, make_response
from flask_cors import CORS
from rpi5_ws2812.ws2812 import Color, WS2812SpiDriver

app = Flask(__name__)
CORS(app, supports_credentials=True)

strip = WS2812SpiDriver(spi_bus=0, spi_device=0, led_count=60).get_strip()

@app.after_request
def add_cors_headers(response):
    # Standard-CORS Header
    response.headers["Access-Control-Allow-Origin"] = "https://one.beezlebug.com"
    response.headers["Access-Control-Allow-Methods"] = "GET, OPTIONS"
    response.headers["Access-Control-Allow-Headers"] = "Content-Type"
    response.headers["Access-Control-Allow-Credentials"] = "true"
    response.headers["Access-Control-Allow-Private-Network"] = "true"
    return response

@app.route("/color")
def color():
    r = int(request.args.get("r", 0))
    g = int(request.args.get("g", 0))
    b = int(request.args.get("b", 0))
    strip.set_all_pixels(Color(r, g, b))
    strip.show()
    return jsonify({"ok": True, "color": [r, g, b]})

if __name__ == "__main__":
    app.run(
        host="0.0.0.0",
        port=5000,
        ssl_context=("certs/cert.pem", "certs/key.pem")
    )
